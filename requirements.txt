# Taipei Metro 1:1 Minecraft rebuild: direct dependencies
#
# Lists only the packages the project's own source (mrt/, cli/, tools/, tests/)
# imports, pinned to the versions that run in the current development
# environment. affine, certifi, click and attrs come in under rasterio; they are
# not listed here and are left to pip to resolve.
#
# Set up the environment (the venv is Python 3.9; the code uses no 3.10+ syntax):
#
#     python3 -m venv .venv
#     ./.venv/bin/pip install -r requirements.txt
#
# After that, always run commands with ./.venv/bin/python, never the system python.

# --- World generation core ---
numpy==2.0.2          # Elevation grid, block arrays and alignment math; used in almost every layer
nbtlib==2.0.4         # Reads and writes the NBT in Anvil world saves (infrastructure/mcworld.py and three verifiers in tools/)

# --- Data pipeline ---
pyproj==3.6.1         # WGS84 -> TWD97 projection; needed by adapters/projection.py, fetch_details and make_heightmap

# --- DEM stage only ---
# Builds data/heightmap.npy from the Ministry of the Interior GeoTIFF
# (mrt.adapters.dem.make_heightmap). The elevation grid is computed once and
# saved as .npy; world generation afterwards only reads it back with numpy, so
# this package can be skipped if you don't rerun this step.
rasterio==1.4.3

# --- Verifiers only ---
# tools/verify_render.py draws a read-back world save as a top-down PNG.
# Generation itself does not use it.
pillow==11.3.0
